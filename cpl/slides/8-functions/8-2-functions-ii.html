回顾: 形参、实参与按值传递
代码
void f(int n) { n = n + 1; }        /* 改的是副本 */
void swap(int a, int b) { int t = a; a = b; b = t; }

int x = 10, y = 5;
f(x);        /* x 仍是 10 */
swap(x, y);  /* x、y 都没变 */
要点

  • 形式参数: 定义里的占位名; 进入函数体就是一个已初始化的局部变量 (拿到实参的副本)

  • 实际参数: 调用时传入的值——常量、变量或任意同类型表达式

  • 按值传递: 复制一份给形参, 之后两者互不影响——swap 失效的根因; 相当于把文件复印一份交给函数

  • 数组是例外 (函数能改到原数组)——因为传的是首地址, 下一页展开

程序: 判定素数 · 函数版
函数版（main 只剩输入输出）
#include <stdio.h>
int is_prime(int n) {
  if (n <= 1) return 0;
  for (int d = 2; d <= n / d; d++)
    if (n % d == 0) return 0;
  return 1;
}
int main(void) {
  int n;
  printf("输入一个数: ");
  scanf("%d", &n);
  if (is_prime(n)) printf("是素数\n");
  else printf("不是素数\n");
  return 0;
}
要点

  • divisor <= n / divisor 是 \(\sqrt{n}\) 的等价写法 (不用 math.h), 不会溢出; 原版的 d * d 写法在 n 近 int 上限时溢出 (有符号溢出是未定义行为)——结果为负时条件恒真, 循环停不下来

  • 顺手补上 n <= 1——原版对 n = 1 会误判; 改造不只是搬代码, 也顺手修 bug

  • 返回 1/0, if (is_prime(n)) 直接当条件——与 8-1 一致

改造进阶: return 让代码更简洁
代码
/* 啰嗦版: 用 if 包装 */
int is_even(int n) {
  if (n % 2 == 0) {
    return 1;
  }
  return 0;
}
/* 简洁版: 布尔表达式直接 return */
int is_even(int n) { return n % 2 == 0; }
int is_digit(char c) { return c >= '0' && c <= '9'; }  /* 数字 */
double average(double a, double b) { return (a + b) / 2; }  /* 两个参数, 照样一行 */
要点

  • 判断的结果本身就是 1 或 0——不需要 if 包装, 直接 return 表达式

  • 一类经典小函数都是 "一行函数": 偶数、数字、大小写字母……参数也可以有多个——average(a, b) 两个参数照样一行 return

  • 提前 return 也让代码更简洁: 找到就返回, 不需要 flag 变量——is_prime 就是这么写的

primes.c: 函数版 
/* 函数版: is_prime + count_primes 两层 */
#include <stdio.h>

int is_prime(int n) {
  if (n <= 1) {
    return 0;
  }
  for (int j = 2; j <= n / j; j++) {
    if (n % j == 0) {
      return 0;
    }
  }
  return 1;
}

int count_primes(int max) {        /* 统计也抽成函数 */
  int count = 0, i;

  for (i = 2; i <= max; i++) {
    if (is_prime(i) == 1) {        /* 调用 is_prime */
      count++;
    }
  }
  return count;
}

int main() {
  int max;
  scanf("%d", &max);
  printf("\n素数个数: %d", count_primes(max));   /* 一行调用 */
  return 0;
}

  • 两层调用链: main → count_primes → is_prime——函数不光能被 main 调用, 还能调用别的函数

  • is_prime 被复用了max-1 次 (i = 2..max), 与前面判定素数的函数是同一个——写一次, 处处用

  • main 只剩读入和一行输出——主函数开始像一份 "说明书": 只写步骤, 细节藏进函数

一维数组作参数: 开头位置与长度
函数要操作数组, 就把数组传给函数:
/* 求和: 函数版 */
#include <stdio.h>
int sum_array(int a[], int n)    /* 形参: 数组+长度 */
{
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
  }
  return sum;
}
int main(void)
{
  int b[5] = {3, 7, 1, 9, 4};
  int c[3] = {10, 20, 30};
  printf("%d\n", sum_array(b, 5));   /* 调用: 只写数组名 */
  printf("%d\n", sum_array(c, 3));   /* 换一组数据 */
  return 0;
}

sum-array-re.c (改造前) · sum-array-func.c (函数版)

  • 形参 int a[] 只拿到首元素地址: 在表达式里 a ≡ &a[0], 不携带长度——长度必须由调用者单独传 n

  • a[i] ≡ *(a + i): 从首地址往后数 i 个——所以改 a[i] 就是改 b[i]; 函数里改数组 = 改原数组

  • 因此 int a[] 形参与 int *a 等价; *、& 的细节在指针章节继续展开

int sum_matrix(int a[][4], int rows)   /* 列数 4 不能省 */
{
  int sum = 0;

要点

  • 二维数组作参数, 只能省略第一维——列数必须写出来

  • 为什么列数不能省? 下一页用内存图细讲——先记住 a[][4] 的 4 不能省

  • 调用: 传数组名 + 行数——写法朴素, 但数组传的是开头位置, 函数能改到原数组 (和标量不同)

sum-matrix-re.c (改造前) · sum-matrix-func.c (函数版)
二维数组在内存里: 行主序

  • 二维数组在内存里按行连续存放: 第 0 行存完, 紧接着第 1 行、第 2 行……

  • 要定位 a[i][j], 先跳过 i 行, 再走 j 个: 偏移 = i × 列数 + j——这是相对 a (即 a[0][0]) 的元素偏移, 乘上 sizeof(int) 才是字节偏移

  • "列数"就是行与行之间的步长——形参 a[][4] 的 4 不能省: 编译器要用它算 a[i][j] 的地址

  • 反过来: 行数不影响步长, 所以第一维可以省略

生命游戏 · 原版 
/* 演化一代: 数邻居 + 套规则 + 打印, 全在 main 里 */
#include <stdio.h>
#define SIZE 8

int main(void)
{
  int board[SIZE + 2][SIZE + 2] = {{0}};
  int next[SIZE + 2][SIZE + 2] = {{0}};
  /* 初始: 一个滑翔机 (初始画面见右栏) */
  board[2][3] = 1;  board[3][4] = 1;
  board[4][2] = 1;  board[4][3] = 1;  board[4][4] = 1;

  for (int i = 1; i <= SIZE; i++) {
    for (int j = 1; j <= SIZE; j++) {
      int cnt = -board[i][j];
      for (int m = i - 1; m <= i + 1; m++) {
        for (int n = j - 1; n <= j + 1; n++) {
          cnt += board[m][n];
        }
      }
      if (board[i][j] == 1) {
        next[i][j] = (cnt == 2 || cnt == 3);
      } else {
        next[i][j] = (cnt == 3);
      }
    }
  }

  for (int i = 1; i <= SIZE; i++) {    /* 打印新一代 */
    for (int j = 1; j <= SIZE; j++) {
      printf("%c", next[i][j] ? '*' : '.');
    }
    printf("\n");
  }
  return 0;
}

int count_neighbors(int board[][SIZE + 2], int i, int j)
{
  int cnt = -board[i][j];

  for (int m = i - 1; m <= i + 1; m++) {
    for (int n = j - 1; n <= j + 1; n++) {
      cnt += board[m][n];
    }
  }
  return cnt;
}

void next_generation(int board[][SIZE + 2])
{
  int new_board[SIZE + 2][SIZE + 2] = {{0}};

  for (int i = 1; i <= SIZE; i++) {
    for (int j = 1; j <= SIZE; j++) {
      int cnt = count_neighbors(board, i, j);
      if (board[i][j] == 1) {
        new_board[i][j] = (cnt == 2 || cnt == 3);
      } else {
        new_board[i][j] = (cnt == 3);
      }
    }
  }
  /* 先算后写: 后面的格子还要读旧的邻居状态, 不能边算边写 */
  for (int i = 1; i <= SIZE; i++) {
    for (int j = 1; j <= SIZE; j++) {
      board[i][j] = new_board[i][j];
    }
  }
}

  • count_neighbors: 一个格子数邻居——cnt = -board[i][j] 先把自己减掉, 因为自己也被数进去了

  • next_generation: 对每个格子调用 count_neighbors 并套三条规则, 结果写进 new_board 再写回 board——数组形参让这个写回能改到原数组

  • 为什么两块棋盘? 先全部算到 new_board 再整体写回——本代已改的格子会污染后面格子的邻居判断

  • 函数之间嵌套调用: next_generation 调 count_neighbors——函数可以调用函数

课堂编码: 挑战 · 组合数

挑战: 用今天写好的 factorial 写 combination——C(m,n) = m! / (n!(m-n)!), 原型 long combination(int m, int n); main → combination → factorial 两层调用链, 运行时栈上就叠着刚才那样的两层便签

factorial 被调用了三次、参数各不同——小函数可以组合成更大的函数

追问: 试试 m = 30——为什么结果不对? (提示: 30! 超出 long 的容量——Windows 上 long 只有 32 位, 13! 就溢出; Mac/Linux 64 位也到 20!)
